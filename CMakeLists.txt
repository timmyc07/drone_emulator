cmake_minimum_required(VERSION 3.20)
project(UniversalUAVSimulator VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
add_executable(uavsim_info src/uavsim_info.cpp)
target_include_directories(uavsim_info PRIVATE include)
add_executable(uavsim_smoke tests/smoke_test.cpp)
target_include_directories(uavsim_smoke PRIVATE include)
if(BUILD_TESTING)
  add_test(NAME uavsim_smoke COMMAND uavsim_smoke)
endif()

option(UAVSIM_BUILD_CPP20_PROBE "Build the extended compiler smoke probe" ON)
if(UAVSIM_BUILD_CPP20_PROBE)
  add_executable(uavsim_cpp20_probe src/main.cpp)
  target_include_directories(uavsim_cpp20_probe PRIVATE include)
  if(MSVC)
    target_compile_options(uavsim_cpp20_probe PRIVATE /W4 /WX /Zc:__cplusplus)
  else()
    target_compile_options(uavsim_cpp20_probe PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
  if(BUILD_TESTING)
    add_test(NAME cpp20_smoke COMMAND uavsim_cpp20_probe --self-test)
    add_test(NAME reject_unknown_argument COMMAND uavsim_cpp20_probe --invalid)
    set_tests_properties(reject_unknown_argument PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
