cmake_minimum_required(VERSION 3.20)
project(UniversalUAVSimulator VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
add_executable(uavsim_info src/uavsim_info.cpp)
target_include_directories(uavsim_info PRIVATE include)
add_executable(uavsim_smoke tests/smoke_test.cpp)
target_include_directories(uavsim_smoke PRIVATE include)
if(BUILD_TESTING)
  add_test(NAME uavsim_smoke COMMAND uavsim_smoke)
endif()

# Portable simulation and configuration; rendering remains a separate Unreal plugin.
add_library(uavsim_core STATIC src/simulation.cpp)
target_include_directories(uavsim_core PUBLIC include)
target_include_directories(uavsim_core SYSTEM PUBLIC third_party/eigen)
target_compile_features(uavsim_core PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(uavsim_core PRIVATE /W4 /Zc:__cplusplus)
else()
  target_compile_options(uavsim_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(uavsim_config STATIC src/config.cpp)
target_link_libraries(uavsim_config PUBLIC uavsim_core)
target_include_directories(uavsim_config SYSTEM PRIVATE third_party third_party/picosha2)

add_executable(uavsim_headless src/headless.cpp)
target_link_libraries(uavsim_headless PRIVATE uavsim_config)
target_include_directories(uavsim_headless SYSTEM PRIVATE third_party third_party/picosha2)

if(BUILD_TESTING)
  if(WIN32)
    find_program(UAVSIM_PWSH pwsh REQUIRED)
    add_test(NAME vs_environment_paths COMMAND "${UAVSIM_PWSH}" -NoProfile -File "${CMAKE_SOURCE_DIR}/tools/validation/test_vs_environment.ps1")
  endif()
  add_executable(uavsim_physics_tests tests/physics_tests.cpp)
  target_link_libraries(uavsim_physics_tests PRIVATE uavsim_core)
  add_test(NAME physics_acceptance COMMAND uavsim_physics_tests)
  add_executable(uavsim_config_tests tests/config_tests.cpp)
  target_link_libraries(uavsim_config_tests PRIVATE uavsim_config)
  target_include_directories(uavsim_config_tests SYSTEM PRIVATE third_party third_party/picosha2)
  add_test(NAME config_acceptance COMMAND uavsim_config_tests "${CMAKE_SOURCE_DIR}/configs/vehicles/generic_quadrotor_x01.json")
endif()

option(UAVSIM_BUILD_CPP20_PROBE "Build the extended compiler smoke probe" ON)
if(UAVSIM_BUILD_CPP20_PROBE)
  add_executable(uavsim_cpp20_probe src/main.cpp)
  target_include_directories(uavsim_cpp20_probe PRIVATE include)
  if(MSVC)
    target_compile_options(uavsim_cpp20_probe PRIVATE /W4 /WX /Zc:__cplusplus)
  else()
    target_compile_options(uavsim_cpp20_probe PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
  if(BUILD_TESTING)
    add_test(NAME cpp20_smoke COMMAND uavsim_cpp20_probe --self-test)
    add_test(NAME reject_unknown_argument COMMAND uavsim_cpp20_probe --invalid)
    set_tests_properties(reject_unknown_argument PROPERTIES WILL_FAIL TRUE)
  endif()
endif()
